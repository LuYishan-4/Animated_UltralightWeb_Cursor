cmake_minimum_required(VERSION 3.16)
project(gui_niri)
message(STATUS "Building Niri settings GUI")
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
qt_add_executable(settings_niri main.cpp)
qt_add_qml_module(settings_niri URI settings.niri VERSION 1.0 QML_FILES main.qml)
target_sources(settings_niri PRIVATE SettingsBackend.cpp)
target_include_directories(settings_niri PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(settings_niri PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2)
install(TARGETS settings_niri RUNTIME DESTINATION bin)
