cmake_minimum_required(VERSION 3.16)
project(gui_hyprland)
message(STATUS "Building Hyprland settings GUI")
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
qt_add_executable(settings_hyprland main.cpp)
qt_add_qml_module(settings_hyprland URI settings.hyprland VERSION 1.0 QML_FILES main.qml)
target_sources(settings_hyprland PRIVATE SettingsBackend.cpp)
target_include_directories(settings_hyprland PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(settings_hyprland PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2)
install(TARGETS settings_hyprland RUNTIME DESTINATION bin)
