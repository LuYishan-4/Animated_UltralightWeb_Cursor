cmake_minimum_required(VERSION 3.16)
project(gui_windows)
message(STATUS "Building Windows settings GUI")
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
qt_add_executable(settings_windows main.cpp)
qt_add_qml_module(settings_windows URI settings.windows VERSION 1.0 QML_FILES main.qml)
target_sources(settings_windows PRIVATE SettingsBackend.cpp)
target_include_directories(settings_windows PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(settings_windows PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2)
install(TARGETS settings_windows RUNTIME DESTINATION bin)
